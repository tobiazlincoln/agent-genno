      /**
       * Recover from a stale chunk graph — "Importing a module script failed."
       * in PostHog (iOS Safari, /staking/calculator and /): typically a tab
       * that outlived a deploy asking for a hashed chunk the new HTML no
       * longer references, or a flaky mobile network. Two paths raise it:
       *   - Vite's own `import()` wrapper → `vite:preloadError` (cancelable);
       *   - an Astro island's component-url / renderer-url import, which the
       *     island runtime retries once (`?astro-retry=`) and then reports as
       *     `astro:hydration-error` (bubbles to document, detail.error).
       * Only MODULE-LOAD failures qualify on the island path — a component
       * that throws while rendering must keep surfacing as an error. Reload
       * once per session to pick up the current graph; a second failure is
       * left alone so it is still reported.
       */
      ;(function () {
        var key = 'helius:preload-reloaded'
        function isModuleLoadError(err) {
          var msg = String((err && err.message) || err || '')
          return (
            /Importing a module script failed/i.test(msg) || // Safari
            /Failed to fetch dynamically imported module/i.test(msg) || // Chrome
            /error loading dynamically imported module/i.test(msg) // Firefox
          )
        }
        function recoverOnce(event) {
          try {
            if (sessionStorage.getItem(key)) return
            sessionStorage.setItem(key, '1')
          } catch (e) {
            return
          }
          event.preventDefault()
          window.location.reload()
        }
        window.addEventListener('vite:preloadError', recoverOnce)
        document.addEventListener('astro:hydration-error', function (event) {
          var err = event.detail && event.detail.error
          if (isModuleLoadError(err)) recoverOnce(event)
        })
      })()
